0.0.1 ［補足］「ラプラス変換が存在するsの範囲」も求める場合．

図 1:
(%i1) f(t):= sin(λ·t);
(%o1) f ⁡(t ) := sin ⁡(λ ⁢ t )
(%i2) assume(T>0);
(%o2) [T >0 ]
(%i3) /*ラプラス変換の定義式において，積分の上限∞をTと置きかえた場合*/
LT: integrate(f(t)·%e^(−s·t), t, 0, T) ;
(LT) λ λ 2 + s 2 -s ⁢ sin ⁡(T ⁢ λ ) + λ ⁢ cos ⁡(T ⁢ λ ) %e T ⁢ s ⁢ λ 2 + %e T ⁢ s ⁢ s 2
(%i4) /*上式の第2項目を因数分解して，置きかえる．*/
LT2: substpart(factor(piece), LT, 2);
(LT2) λ λ 2 + s 2 -%e - (T ⁢ s ) ⁢ (s ⁢ sin ⁡(T ⁢ λ ) + λ ⁢ cos ⁡(T ⁢ λ ) ) λ 2 + s 2
--> /*T→∞の場合，s<0だと上式の第2項目の指数関数が∞になるので，積分は収束しない．
また，s=0だと，cos(∞λ)の項が残るため，積分は収束しない．
したがって，ラプラス変換が存在するsの範囲は，s>0となる．【答え】*/;
--> /*一応，LT2の極限を調べると，(s>0の場合の式が，ラプラス変換した式である．)*/;
(%i5) /*s>0, λ>0の場合*/
limit(LT2, T, inf); /*「p」，「p」と答える．*/
Is s positive, negative or zero? Is λ positive, negative or zero?
(%o5) λ λ 2 + s 2
(%i6) /*s>0, λ<0の場合*/
limit(LT2, T, inf); /*「p」，「n」と答える．*/
Is s positive, negative or zero? Is λ positive, negative or zero?
(%o6) λ λ 2 + s 2
(%i7) /*s>0, λ=0の場合*/
limit(LT2, T, inf); /*「p」，「z」と答える．*/
Is s positive, negative or zero? Is λ positive, negative or zero?
(%o7) 0
(%i8) /*s<0, λ>0の場合*/
limit(LT2, T, inf); /*「n」，「p」と答える．*/
Is s positive, negative or zero? Is λ positive, negative or zero?
(%o8) und
--> /*「und」は，undefined(未定義)．∞か-∞になるので，値として定義できないという意味．*/;
(%i9) /*s=0, λ>0の場合*/
limit(LT2, T, inf); /*「z」，「p」と答える．*/
Is s positive, negative or zero? Is λ positive, negative or zero?
(%o9) ind
--> /*「ind」は，値が定まらないという意味．cos(∞λ)の項があるため．*/;
(%i10) limit(LT2, s, 0);
(%o10) 1 λ -cos ⁡(T ⁢ λ ) λ
