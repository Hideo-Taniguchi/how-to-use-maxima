0.0.1 ［補足］「ラプラス変換が存在するsの範囲」も求める場合．

図 1:
(%i1) assume(λ>0) ; /*定数の仮定*/
(%o1) [λ >0 ]
(%i2) /*ヘビサイドの単位関数*/
f(t):= hstep(t−λ);
(%o2) f ⁡(t ) := hstep ⁡(t -λ )
(%i3) /*ヘビサイドの単位関数で，hstep(0+ε)とするための変数．広義積分のため．*/
assume(ε>0);
(%o3) [ε >0 ]
(%i4) assume(T>0);
(%o4) [T >0 ]
(%i5) /*ラプラス変換の定義式において，積分の上限∞をTと置きかえた場合．
今回は，下限をλ+εにしている．*/
LT: integrate(f(t)·%e^(−s·t), t, λ+ε, T) ;
(LT) ∫λ + ε T %e - (s ⁢ t ) ⁢ (1 -hstep ⁡(λ -t ) ) d t
--> /*【Maximaコマンド】
changevar(式, g(x,y), y, x)
式中のxに関する積分を，g(x,y)=0によって，yに関する積分に変換する．
*/;
(%i6) /*変数の置換．τ=t-λ*/
LT2: changevar(LT, t−τ−λ, τ, t); /*積分の変数変換*/
(LT2) ∫ε T -λ %e -(s ⁢ τ ) -s ⁢ λ ⁢ hstep ⁡(τ ) d τ
(%i7) /*被積分関数の内，hstep(τ)は，区間[ε,T-λ]で1なので，1に置きかえる．*/
LT3: substpart(1, LT2, 1, 2);
(LT3) ∫ε T -λ %e -(s ⁢ τ ) -s ⁢ λ d τ
(%i8) /*LT3の積分を強制的に評価する*/
ev(LT3, risch);
(%o8) %e -(s ⁢ λ ) -s ⁢ ε s -%e - (T ⁢ s ) s
(%i9) /*ε→0の極限をとる．*/
LT4: limit(%, ε, 0);
(LT4) %e - (s ⁢ λ ) s -%e - (T ⁢ s ) s
--> /*T→∞の場合，s<0だと上式の第2項目の指数関数が∞になるので，積分は収束しない．
また，s=0だと，0/0になるので，積分は収束しない．
したがって，ラプラス変換が存在するsの範囲は，s>0となる．【答え】*/;
--> /*一応，LT4の極限を調べると，(s>0の場合の式が，ラプラス変換した式である．)*/;
(%i10) /*s>0の場合*/
limit(LT4, T, inf); /*「p」と答えること*/
Is s positive, negative or zero?
(%o10) %e - (s ⁢ λ ) s
(%i11) /*s<0の場合*/
limit(LT4, T, inf); /*「n」と答えること*/
Is s positive, negative or zero?
(%o11) inf
(%i12) /*s=0の場合*/
limit(LT4, T, inf); /*「z」と答えること*/
