ラプラス変換：lapace(式, t, s)
　(式は変数tに関するもの．sはラプラス変換のパラメーター．)
－－－－－－－－－－－－－－－－－－－－－－－－－－－－*/;
--> /*ここでは，Maximaのlaplace関数でラプラス変換を求める方法を示します．
この方法では，「ラプラス変換が存在するsの範囲」が自明であるため，
それを説明することはできません．*/;
図 1:
(%i1) laplace(1, t, s);
(%o1) 1 s
図 2:
(%i2) /*記号の並びのために，あえて宣言する．*/
declare(λ, constant);   /*λを定数として宣言*/
(%o2) done
(%i3) /*【注意】Maximaでは，ヘビサイドの単位関数の時間のズレλについて，
assume(λ>=0)とすると，ラプラス変換において，「IS λ negative or zero?」
と尋ねられ，λ>0の条件が有効にならない．いわゆる，バグ．それにより，
模範解答の答えが得られないので，0を含めずに，assume(λ>0)とする．*/
assume(λ>0) ; /*定数の仮定*/
(%o3) [0 <λ ]
(%i4) laplace(hstep(t−λ), t, s);
(%o4) %e - (λ ⁢ s ) s
図 3:
(%i5) (forget(λ>0), facts(λ)); /*問(2)の「λ>0」の条件を削除し，確認する．*/
(%o5) [kind ⁡(λ , constant ) ]
(%i6) laplace(%e^(λ·t), t, s);
(%o6) 1 s -λ
図 4:
(%i7) assume(λ>−1) ; /*定数の仮定*/
(%o7) [-λ -1 <0 ]
(%i8) laplace(t^λ, t, s);
(%o8) γ ⁡(λ + 1 ) ⁢ s -λ -1
(%i9) declare(n, integer); /*nを整数として宣言*/
(%o9) done
(%i10) assume(n>=0); /*nを0以上と仮定*/
(%o10) [n >= 0 ]
(%i11) laplace(t^n, t, s);
(%o11) γ ⁡(n + 1 ) ⁢ s -n -1
--> /*【コメント】 nが整数で，n≧0だとしても，Maximaでは，Γ(n+1)は，n!には変換されない．*/;
図 5:
(%i12) (forget(λ>−1), facts(λ)); /*問(4)の「λ>-1」の条件を削除し，確認する．*/
(%o12) [kind ⁡(λ , constant ) ]
(%i13) laplace(cos(λ·t), t, s);
(%o13) s s 2 + λ 2

(%i14) laplace(sin(λ·t), t, s);
